74static uint8_t a_sfa40_generate_crc(uint8_t* data, uint8_t count)
75{
76 uint8_t current_byte;
77 uint8_t crc_bit;
78 uint8_t crc = 0xFF;
79
80 for (current_byte = 0; current_byte < count; ++current_byte) /* calculate crc */
81 {
82 crc ^= (data[current_byte]); /* xor data */
83 for (crc_bit = 8; crc_bit > 0; --crc_bit) /* 8 bit */
84 {
85 if ((crc & 0x80) != 0) /* if 7th bit is 1 */
86 {
87 crc = (crc << 1) ^ 0x31; /* xor */
88 }
89 else
90 {
91 crc = crc << 1; /* left shift 1 */
92 }
93 }
94 }
95
96 return crc; /* return crc */
97}
